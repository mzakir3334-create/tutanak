var OKUL_KLASORU = 'Yaşar Doğu Ortaokulu';

var SAAT_DILIMI = 'Europe/Istanbul';
var EN_COK_HATALI_DENEME = 20; // 10 dakikada
var KILIT_SANIYE = 600;
var EN_COK_FOTO = 3;
var EN_COK_FOTO_BAYT = 2500000;

function doGet(e) {
  var kod = '';
  try { kod = String((e && e.parameter && e.parameter.k) || '').toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 20); } catch (x) { /* yok say */ }
  var html = SAYFA_HTML.replace('__KOD__', kod);
  return HtmlService.createHtmlOutput(html)
    .setTitle('Tutanak Bildirimi')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

var IZINLI_ISLEVLER_ = { giris: function (a) { return giris.apply(null, a); }, ara: function (a) { return ara.apply(null, a); }, gonder: function (a) { return gonder.apply(null, a); }, benimkiler: function (a) { return benimkiler.apply(null, a); }, adTara: function (a) { return adTara.apply(null, a); } };

/** Telefon uygulamasından gelen istek: gövde JSON { fn, args }. Yalnızca beş işleve izin verilir; kod doğrulaması işlevlerin içinde yapılır. */
function doPost(e) {
  var cevap;
  try {
    var istek = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    var fn = String(istek.fn || '');
    if (!IZINLI_ISLEVLER_[fn]) throw new Error('Geçersiz istek.');
    var args = Array.isArray(istek.args) ? istek.args : [];
    var v = IZINLI_ISLEVLER_[fn](args);
    cevap = { ok: true, v: v };
  } catch (x) {
    cevap = { ok: false, hata: String((x && x.message) || x) };
  }
  return ContentService.createTextOutput(JSON.stringify(cevap)).setMimeType(ContentService.MimeType.JSON);
}

// ------------------------------------------------------------------ Yardımcılar

function norm_(s) {
  return String(s || '')
    .replace(/İ/g, 'i').replace(/I/g, 'ı').toLowerCase()
    .replace(/ı/g, 'i').replace(/ğ/g, 'g').replace(/ü/g, 'u').replace(/ş/g, 's').replace(/ö/g, 'o').replace(/ç/g, 'c')
    .replace(/[^a-z0-9]+/g, ' ').trim();
}

function klasorAltinda_(ust, ad, olustur) {
  var it = ust.getFolders();
  var hedef = norm_(ad);
  while (it.hasNext()) {
    var f = it.next();
    if (norm_(f.getName()) === hedef) return f;
  }
  return olustur ? ust.createFolder(ad) : null;
}

function dosyaBul_(klasor, ad) {
  var it = klasor.getFilesByName(ad);
  var en = null;
  while (it.hasNext()) {
    var f = it.next();
    if (f.isTrashed()) continue;
    if (!en || f.getLastUpdated() > en.getLastUpdated()) en = f;
  }
  return en;
}

/** Veri klasörü: kayıtlı kimlik; yoksa "Disiplin Kurulu" adlı (içinde oddk.db olan) klasör aranır */
function klasor_() {
  var props = PropertiesService.getScriptProperties();
  var id = props.getProperty('KLASOR_ID');
  if (id) {
    try { var f = DriveApp.getFolderById(id); if (!f.isTrashed()) return f; } catch (e) { /* aranacak */ }
  }
  var gorulen = {}, en = null, enPuan = -1;
  ['Disiplin', 'DİSİPLİN', 'DISIPLIN', 'disiplin'].forEach(function (w) {
    var it = DriveApp.searchFolders("title contains '" + w + "' and trashed = false and 'me' in owners");
    while (it.hasNext()) {
      var k = it.next();
      if (gorulen[k.getId()] || norm_(k.getName()) !== norm_(VERI_KLASORU)) continue;
      gorulen[k.getId()] = true;
      var puan = 0;
      var ustler = k.getParents();
      while (ustler.hasNext()) if (norm_(ustler.next().getName()).indexOf(norm_(OKUL_KLASORU).split(' ').slice(0, 2).join(' ')) >= 0) puan += 2;
      if (k.getFilesByName('oddk.db').hasNext()) puan += 1;
      if (puan > enPuan) { en = k; enPuan = puan; }
    }
  });
  if (!en) throw new Error('Veri klasörü bulunamadı. ÖDDK programını bir kez açın, sonra kurulumu yeniden deneyin.');
  props.setProperty('KLASOR_ID', en.getId());
  return en;
}

function ozet_(kod, tuz) {
  var bayt = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, tuz + kod, Utilities.Charset.UTF_8);
  return bayt.map(function (b) { return ('0' + (b < 0 ? b + 256 : b).toString(16)).slice(-2); }).join('');
}

/** Programın yayımladığı veri dosyası (öğretmenler + kod özetleri, öğrenciler, davranış listesi). Kısa süre önbelleğe alınır. */
function veri_() {
  var cache = CacheService.getScriptCache();
  var n = cache.get('v_n');
  if (n) {
    var parca = [];
    for (var i = 0; i < Number(n); i++) { var p = cache.get('v_' + i); if (p === null) { parca = null; break; } parca.push(p); }
    if (parca) { try { return JSON.parse(parca.join('')); } catch (e) { /* yeniden oku */ } }
  }
  var k = klasorAltinda_(klasor_(), SAYFA_KLASORU, false);
  var f = k && dosyaBul_(k, SAYFA_DOSYASI);
  if (!f) throw new Error('Öğretmen sayfası henüz yayınlanmamış. ÖDDK programında Ayarlar › Öğretmen Telefon Sayfası bölümünden "Güncelle"ye basın.');
  var metin = f.getBlob().getDataAsString('UTF-8');
  try {
    var adet = Math.ceil(metin.length / 90000);
    for (var j = 0; j < adet; j++) cache.put('v_' + j, metin.slice(j * 90000, (j + 1) * 90000), 120);
    cache.put('v_n', String(adet), 120);
  } catch (e2) { /* önbellek yazılamazsa her seferinde okunur */ }
  return JSON.parse(metin);
}

/** Kodu doğrular. Başarılıysa { v: veri, o: öğretmen }; değilse hata fırlatır (çok hatalı denemede kilitler). */
function dogrula_(kod) {
  var cache = CacheService.getScriptCache();
  var hatali = Number(cache.get('hatali') || 0);
  if (hatali >= EN_COK_HATALI_DENEME) throw new Error('Çok fazla hatalı deneme yapıldı. Lütfen 10 dakika sonra tekrar deneyin.');
  var temiz = String(kod || '').toUpperCase().replace(/[^A-Z0-9]/g, '');
  var v = veri_();
  var o = null;
  if (temiz.length >= 6) {
    var oz = ozet_(temiz, v.tuz);
    for (var i = 0; i < (v.ogretmenler || []).length; i++) if (v.ogretmenler[i].ozet && v.ogretmenler[i].ozet === oz) { o = v.ogretmenler[i]; break; }
  }
  if (!o) {
    cache.put('hatali', String(hatali + 1), KILIT_SANIYE);
    throw new Error('Kod hatalı. Kodunuzu müdür yardımcısından öğrenebilirsiniz.');
  }
  return { v: v, o: o };
}

function bugun_() { return Utilities.formatDate(new Date(), SAAT_DILIMI, 'yyyy-MM-dd'); }

// ------------------------------------------------------------------ Sayfanın çağırdığı işlevler

function giris(kod) {
  var d = dogrula_(kod);
  return {
    ad: d.o.ad,
    brans: d.o.brans || '',
    okul: d.v.okul || '',
    ogretmenler: (d.v.ogretmenler || []).map(function (x) { return { id: x.id, ad: x.ad, brans: x.brans || '' }; }),
    davranislar: d.v.davranislar || [],
    bugun: bugun_(),
  };
}

/** Öğrenci arama: en az 2 harf; toplu liste verilmez, en çok 8 eşleşme */
function ara(kod, q) {
  var d = dogrula_(kod);
  var kelimeler = norm_(q).split(' ').filter(function (x) { return x; });
  if (!kelimeler.length || norm_(q).length < 2) return [];
  var out = [];
  var l = d.v.ogrenciler || [];
  for (var i = 0; i < l.length && out.length < 8; i++) {
    var s = l[i];
    var havuz = norm_(s.ad + ' ' + (s.sinif || '') + (s.sube || '') + ' ' + (s.sinif || '') + ' ' + (s.sube || '') + ' ' + (s.no || ''));
    var hepsi = true;
    for (var j = 0; j < kelimeler.length; j++) if (havuz.indexOf(kelimeler[j]) < 0) { hepsi = false; break; }
    if (hepsi) out.push({ id: s.id, ad: s.ad, sinif: s.sinif || '', sube: s.sube || '', no: s.no || '' });
  }
  return out;
}

/**
 * Açıklama metninde geçen öğrenci adlarını bulur. "Ad Soyad" veya "Soyad Ad" biçimi aranır
 * (ek almış hâlleri de: "Ali Azizi'ye"). Tek başına ad eşleştirilmez; yanlış kişi yazılmasın.
 * Dönüş: [{ bas, bit, ek, adaylar: [{id,ad,sinif,sube,no}] }]; bas/bit metindeki karakter aralığı, ek = sondaki ek ("'ye").
 */
function adTara(kod, metin) {
  var d = dogrula_(kod);
  metin = String(metin || '').slice(0, 3000);
  var kelimeler = [], re = /[0-9A-Za-zÇĞİÖŞÜçğıöşüÂâÎîÛû]+(?:['’][A-Za-zÇĞİÖŞÜçğıöşü]+)?/g, m;
  while ((m = re.exec(metin)) !== null) {
    var ap = m[0].search(/['’]/);
    var govde = ap >= 0 ? m[0].slice(0, ap) : m[0];
    kelimeler.push({ bas: m.index, bit: m.index + m[0].length, n: norm_(govde).replace(/ /g, ''), ap: ap >= 0, w: m[0], gl: govde.length });
  }
  if (kelimeler.length < 2) return [];
  // kelime w, jetonla eşleşir mi? (ekli hâl: apostrofsuz ise en çok 5 harf fazla)
  var uyar = function (k, t, ekli) {
    if (k.n === t) return true;
    return !!ekli && !k.ap && t.length >= 3 && k.n.indexOf(t) === 0 && k.n.length - t.length <= 5;
  };
  var grup = {}, l = d.v.ogrenciler || [];
  for (var i = 0; i < l.length; i++) {
    var jet = norm_(l[i].ad).split(' ').filter(function (x) { return x; });
    if (jet.length < 2) continue;
    var soyad = jet[jet.length - 1], adlar = jet.slice(0, -1);
    var buldu = function (a, z, sonJeton) {
      var son = kelimeler[z], ek = '';
      if (son.ap) ek = son.w.slice(son.gl); else if (son.n.length > sonJeton.length) ek = son.w.slice(sonJeton.length);
      var bit = son.bit - ek.length + (son.ap ? 0 : 0);
      // ek kelimenin sonunda: ap'li hâlde bit kelime sonu, ek ayrı
      var anahtar = kelimeler[a].bas + '-' + son.bit;
      var devam = metin.slice(son.bit);
      if (/^\s*\(\s*\d/.test(devam)) return; // sınıfı zaten yazılmış
      if (!grup[anahtar]) grup[anahtar] = { bas: kelimeler[a].bas, bit: son.bit, ek: ek, adaylar: [] };
      grup[anahtar].adaylar.push({ id: l[i].id, ad: l[i].ad, sinif: l[i].sinif || '', sube: l[i].sube || '', no: l[i].no || '' });
    };
    for (var w = 0; w < kelimeler.length; w++) {
      // Ad (…) Soyad: soyad kelimesi, hemen öncesinde ad jetonlarından biri
      if (w > 0 && uyar(kelimeler[w], soyad, true)) {
        var bas = w - 1;
        if (adlar.indexOf(kelimeler[bas].n) >= 0) {
          while (bas - 1 >= 0 && w - bas < adlar.length && adlar.indexOf(kelimeler[bas - 1].n) >= 0) bas--;
          buldu(bas, w, soyad);
          continue;
        }
      }
      // Soyad Ad: soyad kelimesi, hemen ardından ad jetonu
      if (w + 1 < kelimeler.length && kelimeler[w].n === soyad) {
        for (var q = 0; q < adlar.length; q++) {
          if (uyar(kelimeler[w + 1], adlar[q], true)) { buldu(w, w + 1, adlar[q]); break; }
        }
      }
    }
  }
  var liste = Object.keys(grup).map(function (k) { return grup[k]; }).sort(function (a, b) { return a.bas - b.bas || (b.bit - b.bas) - (a.bit - a.bas); });
  var son = [], sonBit = -1;
  liste.forEach(function (g) { if (g.bas >= sonBit && son.length < 10) { son.push(g); sonBit = g.bit; } });
  return son;
}

/**
 * Tutanak bildirimi gönderir. k: { no, ogrenciId, tarih(yyyy-MM-dd), yer, maddeId, aciklama,
 *   fotograflar: [{ base64, mime }] }
 * "no" telefonun ürettiği benzersiz numaradır: aynı kayıt iki kez gönderilirse bir kez yazılır.
 */
function gonder(kod, k) {
  var d = dogrula_(kod);
  k = k || {};
  var no = String(k.no || '').replace(/[^A-Za-z0-9]/g, '').slice(0, 40);
  if (no.length < 8) throw new Error('Kayıt numarası geçersiz, sayfayı yenileyip tekrar deneyin.');
  var ogr = null;
  var l = d.v.ogrenciler || [];
  for (var i = 0; i < l.length; i++) if (String(l[i].id) === String(k.ogrenciId)) { ogr = l[i]; break; }
  if (!ogr) throw new Error('Öğrenci seçilmedi.');
  var tarih = String(k.tarih || '');
  if (!/^\d{4}-\d{2}-\d{2}$/.test(tarih)) throw new Error('Olay tarihini seçin.');
  if (tarih > bugun_()) throw new Error('Olay tarihi ileri bir tarih olamaz.');
  var aciklama = String(k.aciklama || '').replace(/\s+/g, ' ').trim();
  if (aciklama.length < 10) throw new Error('Olay açıklamasını biraz daha ayrıntılı yazın (en az 10 karakter).');
  if (aciklama.length > 3000) throw new Error('Olay açıklaması çok uzun (en fazla 3000 karakter).');
  var yer = String(k.yer || '').trim().slice(0, 80);
  var maddeId = null;
  if (k.maddeId) {
    var dl = d.v.davranislar || [];
    for (var m = 0; m < dl.length; m++) if (String(dl[m].id) === String(k.maddeId)) { maddeId = dl[m].id; break; }
  }
  var fotolar = (k.fotograflar || []).slice(0, EN_COK_FOTO);
  // Yanındaki (nöbetçi) öğretmenler: tutanakta imza yeri açılır; en çok 3
  var ekImzalar = [];
  (Array.isArray(k.ekImzalar) ? k.ekImzalar : []).forEach(function (x) {
    var ad = String((x && x.ad) || '').replace(/\s+/g, ' ').trim().slice(0, 80);
    if (ad.length < 3 || ad === d.o.ad || ekImzalar.length >= 3) return;
    for (var q = 0; q < ekImzalar.length; q++) if (ekImzalar[q].ad === ad) return;
    ekImzalar.push({ ad: ad, unvan: 'Nöbetçi Öğretmen' });
  });

  var kilit = LockService.getScriptLock();
  kilit.waitLock(20000);
  try {
    var gelen = klasorAltinda_(klasor_(), GELEN_KLASORU, true);
    var ad = 'TUT_T' + d.o.id + '_' + no + '.json';
    var islenen = klasorAltinda_(gelen, ISLENEN_KLASORU, false);
    if (dosyaBul_(gelen, ad) || (islenen && dosyaBul_(islenen, ad))) return { ok: true, tekrar: true };
    var fotoAdlari = [];
    for (var f = 0; f < fotolar.length; f++) {
      var mime = /^image\/(jpeg|png|webp|heic|heif)$/.test(String(fotolar[f].mime)) ? String(fotolar[f].mime) : 'image/jpeg';
      var bayt = Utilities.base64Decode(String(fotolar[f].base64 || ''));
      if (!bayt.length) continue;
      if (bayt.length > EN_COK_FOTO_BAYT) throw new Error('Fotoğraf çok büyük, daha küçük bir fotoğraf seçin.');
      var uz = mime === 'image/png' ? '.png' : mime === 'image/webp' ? '.webp' : mime === 'image/jpeg' ? '.jpg' : '.heic';
      var fad = 'TUT_T' + d.o.id + '_' + no + '_f' + (f + 1) + uz;
      gelen.createFile(Utilities.newBlob(bayt, mime, fad));
      fotoAdlari.push(fad);
    }
    var kayit = {
      surum: 1, no: no, ogretmenId: d.o.id, ogretmenAd: d.o.ad,
      ogrenciId: ogr.id, ogrenciAd: ogr.ad, sinifSube: [ogr.sinif, ogr.sube].filter(function (x) { return x; }).join('/'),
      tarih: tarih, yer: yer, maddeId: maddeId, aciklama: aciklama, ekImzalar: ekImzalar, fotograflar: fotoAdlari,
      gonderim: Utilities.formatDate(new Date(), SAAT_DILIMI, "yyyy-MM-dd'T'HH:mm:ss"),
    };
    gelen.createFile(ad, JSON.stringify(kayit), 'application/json'); // JSON en son: fotoğraflar tamamsa kayıt görünür
    return { ok: true };
  } finally {
    kilit.releaseLock();
  }
}

/** Öğretmenin son gönderdikleri ve durumları */
function benimkiler(kod) {
  var d = dogrula_(kod);
  var onEk = 'TUT_T' + d.o.id + '_';
  var gelen = klasorAltinda_(klasor_(), GELEN_KLASORU, false);
  var out = [];
  if (!gelen) return out;
  var topla = function (klasor, durum) {
    var it = klasor.getFiles();
    while (it.hasNext()) {
      var f = it.next();
      var a = f.getName();
      if (f.isTrashed() || a.indexOf(onEk) !== 0 || !/\.json$/.test(a)) continue;
      try {
        var j = JSON.parse(f.getBlob().getDataAsString('UTF-8'));
        out.push({ ogrenci: j.ogrenciAd, sinif: j.sinifSube, tarih: j.tarih, gonderim: j.gonderim, durum: durum });
      } catch (e) { /* bozuk dosya atlanır */ }
    }
  };
  topla(gelen, 'Gönderildi');
  var islenen = klasorAltinda_(gelen, ISLENEN_KLASORU, false);
  if (islenen) topla(islenen, 'İşleme alındı');
  out.sort(function (a, b) { return String(b.gonderim).localeCompare(String(a.gonderim)); });
  return out.slice(0, 15);
}

/** Kurulum sonrası düzenleyicide çalıştırıp günlüğe bakın */
function kurulumuTestEt() {
  var k = klasor_();
  Logger.log('Veri klasörü: ' + k.getName() + ' (' + k.getUrl() + ')');
  var v = veri_();
  Logger.log('Yayınlanan öğretmen: ' + (v.ogretmenler || []).length + ', öğrenci: ' + (v.ogrenciler || []).length + ', kodu olan öğretmen: ' +
    (v.ogretmenler || []).filter(function (o) { return o.ozet; }).length);
  Logger.log('Web uygulaması adresi: ' + ScriptApp.getService().getUrl());
}

/** Veri klasörü yanlış bulunursa: Drive klasör bağlantısını verin (örn. klasorAyarla('https://drive.google.com/drive/folders/XXXX')) */
function klasorAyarla(baglanti) {
  var m = /[-\w]{25,}/.exec(String(baglanti || ''));
  if (!m) throw new Error('Geçerli bir Drive klasör bağlantısı verin');
  var f = DriveApp.getFolderById(m[0]);
  PropertiesService.getScriptProperties().setProperty('KLASOR_ID', f.getId());
  return f.getName();
}

  function kucult(dosya) {
    return new Promise(function (ok) {
      var r = new FileReader();
      r.onload = function () {
        var img = new Image();
        img.onload = function () {
          var m = 1400, s = Math.min(1, m / Math.max(img.width, img.height));
          var c = document.createElement('canvas'); c.width = Math.round(img.width * s); c.height = Math.round(img.height * s);
          c.getContext('2d').drawImage(img, 0, 0, c.width, c.height);
          ok(c.toDataURL('image/jpeg', 0.8));
        };
        img.onerror = function () { ok(null); };
        img.src = r.result;
      };
      r.onerror = function () { ok(null); };
      r.readAsDataURL(dosya);
    });
  }
